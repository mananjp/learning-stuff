#include <iostream>
using namespace std;

class rect {
private:
    int l;
    int b;

public:
    rect(int L, int B)
    {
        l=L;
        b=B;
    }
  

    int area() {
        return l * b;
    }

    int per() {
        return 2 * (l + b);
    }
};

int main() {
    rect r1 = rect(2,5);

    int a = r1.area(); // Calculate area
    int p = r1.per();  // Calculate perimeter

    cout << "Area is: " << a << endl;
    cout << "Perimeter is: " << p << endl;

    return 0;
}
