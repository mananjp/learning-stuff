#include <iostream>
#include <vector>
using namespace std;

int sum(vector<int> vec)
{
    int a = vec.back();
    vec.pop_back();
    if(vec.empty()==1)
    {
        return 0;
    }
    else {
        return a+sum(vec);
    }
    
}


int main()
{
    
    vector<int> vec;
    for(int i=0; i<5; i++)
    {
        vec.push_back(i);
    }

    int x = sum(vec);
    cout<< x <<endl;
    
   
}
