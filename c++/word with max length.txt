#include <iostream>
#include <sstream>
#include <string>

using namespace std;

int main() {
    string input;
    getline(cin,input);
    
    stringstream ss(input);
    string word, longest;

    while (ss >> word) {
        if (word.length() > longest.length()) {
            longest = word;
        }
    }

    cout << "Longest word: " << longest << endl;
    cout<<longest.length();
    return 0;
}
