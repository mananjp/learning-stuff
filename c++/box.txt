#include<bits/stdc++.h>

using namespace std;
class Box {
private:
    int len = 0;
    int wid = 0;
    int hei = 0;

public:
    // Default Constructor (needed for Box temp;)
    Box() {}

    // Parameterized Constructor
    Box(int l, int b, int h) : len(l), wid(b), hei(h) {}

    // Getters
    int getBreadth() { return wid; }
    int getLength() { return len; }
    int getHeight() { return hei; }

    // Calculate Volume (prevent integer overflow)
    long long CalculateVolume() {
        return (long long)len * wid * hei;
    }

    // Overloading '<' Operator
    bool operator<(const Box &b) const {
        if (len < b.len) return true;
        if (len == b.len && wid < b.wid) return true;
        if (len == b.len && wid == b.wid && hei < b.hei) return true;
        return false;
    }

    // Overloading Assignment Operator (needed for temp = NewBox;)
    Box& operator=(const Box &b) {
        if (this != &b) { // Avoid self-assignment
            len = b.len;
            wid = b.wid;
            hei = b.hei;
        }
        return *this;
    }

    // Overloading '<<' for Printing (needed for cout << temp;)
    friend ostream &operator<<(ostream &out, const Box &b) {
        out << b.len << " " << b.wid << " " << b.hei;
        return out;
    }
};



void check2()
