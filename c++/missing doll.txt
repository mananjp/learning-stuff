#include <iostream>
#include <unordered_map>
using namespace std;

int findMissingDoll(int arr[], int n) {
    unordered_map<int, int> dollMap;
    for (int i = 0; i < n; i++) {
        dollMap[arr[i]]++;
    }
    for (auto& doll : dollMap) {
        if (doll.second % 2 != 0) {
            return doll.first;
        }
    }
    return -1; 
}

int main() {
    int t;
    cin >> t;
    while (t--) {
        int n;
        cin >> n;
        int arr[n];
        for (int i = 0; i < n; i++) {
            cin >> arr[i];
        }
        int missingDoll = findMissingDoll(arr, n);
        cout << missingDoll << endl;
    }
    return 0;
}
