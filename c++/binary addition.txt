class Solution {
public:
    string addBinary(string a, string b) {
         int len1 = a.length(), len2 = b.length();
    int maxLen = max(len1, len2);

    // Pad the shorter string with leading zeros to make both equal length
    while (a.length() < maxLen) a = '0' + a;
    while (b.length() < maxLen) b = '0' + b;

    string result = "";  // To store the final sum
    int carry = 0;

    // Traverse from rightmost bit (least significant) to left
    for (int i = maxLen - 1; i >= 0; i--) {
        int bit1 = a[i] - '0';  // Convert character to integer
        int bit2 = b[i] - '0';

        int sum = bit1 + bit2 + carry;  // Add current bits and carry

        // Write the sum bit to the result
        result += (sum % 2) + '0';  // If sum is 2 or 0 → 0; if sum is 1 or 3 → 1
        carry = sum / 2;  // Update carry (1 if sum is 2 or 3)
    }

    // If there's a leftover carry, append it
    if (carry) result += '1';

    // Reverse the result to get the correct order
    reverse(result.begin(), result.end());
    
    return result;
    }
};
