#include <cmath>
#include <cstdio>
#include <vector>
#include <iostream>
#include <algorithm>
using namespace std;


int main() {
int n;
cin>>n;
int q;
cin>>q;
vector<vector<int>> arr(n);
for(int i=0;i<n;i++)
{
    int size;
    cin>>size;
    for(int j=0;j<size;j++)
    {
        int num;
        cin>>num;
        arr[i].push_back(num);
    }
}
for(int i=0;i<q;i++)
{
    int a,b;
    cin>>a>>b;
    cout<<arr[a][b]<<endl;
}   
    return 0;
}
