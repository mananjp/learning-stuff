#include <iostream>
#include <vector>
#include <list> // For adjacency list representation

class Graph {
private:
    int V; // Number of vertices
    std::vector<std::list<int>> adj; // Adjacency list

    // Recursive helper function for DFS traversal
    void DFSUtil(int v, std::vector<bool>& visited) {
        visited[v] = true; // Mark the current node as visited
        std::cout << v << " "; // Process the current node (e.g., print it)

        // Recur for all the vertices adjacent to this vertex
        for (int neighbor : adj[v]) {
            if (!visited[neighbor]) {
                DFSUtil(neighbor, visited); // Recur for unvisited neighbors
            }
        }
    }

public:
    // Constructor to initialize the graph with V vertices
    Graph(int V) {
        this->V = V;
        adj.resize(V); // Resize adjacency list to hold V lists
    }

    // Function to add an edge to the graph
    void addEdge(int v, int w) {
        adj[v].push_back(w); // Add w to v's list (directed graph)
        // For an undirected graph, also add: adj[w].push_back(v);
    }

    // DFS traversal starting from a given vertex
    void DFS(int startVertex) {
        std::vector<bool> visited(V, false); // Mark all vertices as not visited initially
        DFSUtil(startVertex, visited); // Call the recursive helper function
    }
};

int main() {
    // Create a graph with 4 vertices
    Graph g(4);

    // Add edges to the graph (example of a directed graph)
    g.addEdge(0, 1);
    g.addEdge(0, 2);
    g.addEdge(1, 2);
    g.addEdge(2, 0);
    g.addEdge(2, 3);
    g.addEdge(3, 3);

    // Perform DFS starting from vertex 2
    std::cout << "Depth First Traversal (starting from vertex 2): ";
    g.DFS(2);
    std::cout << std::endl;

    return 0;
}
